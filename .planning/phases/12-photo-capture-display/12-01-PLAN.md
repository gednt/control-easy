@Injectable({ providedIn: 'root' })
export class PhotosApiService {
  private http = inject(HttpClient);

  upload(blob: Blob, entityType: PhotoEntityType, entityId: string): Observable<PhotoResponse> {
    const formData = new FormData();
    formData.append('file', blob, 'photo.jpg');
    formData.append('entityType', entityType);
    formData.append('entityId', entityId);
    return this.http.post<PhotoResponse>('/api/v1/photos', formData);
  }

  get(photoId: string): Observable<Blob> {
    return this.http.get(`/api/v1/photos/${photoId}`, { responseType: 'blob' });
  }

  getThumbnail(photoId: string): Observable<Blob> {
    return this.http.get(`/api/v1/photos/${photoId}/thumbnail`, { responseType: 'blob' });
  }

  getThumbnailUrl(photoId: string): string {
    // Direct URL for use in <img src> (avoids blob URL lifecycle)
    return `/api/v1/photos/${photoId}/thumbnail`;
  }

  getSourceUrl(photoId: string): string {
    return `/api/v1/photos/${photoId}`;
  }

  list(entityType: PhotoEntityType, entityId: string): Observable<PhotoResponse[]> {
    return this.http.get<PhotoResponse[]>(`/api/v1/photos?entityType=${entityType}&entityId=${entityId}`);
  }

  delete(photoId: string): Observable<void> {
    return this.http.delete<void>(`/api/v1/photos/${photoId}`);
  }
}

```typescript
export interface CompressionOptions {
  maxDimension: number;        // 1280
  maxBytes: number;            // 500 * 1024
  initialQuality: number;      // 0.8
  fallbackQuality: number;     // 0.6
  floorQuality: number;        // 0.3
  mimeType: 'image/jpeg';
}

export const DEFAULT_COMPRESSION: CompressionOptions = {
  maxDimension: 1280,
  maxBytes: 500 * 1024,
  initialQuality: 0.8,
  fallbackQuality: 0.6,
  floorQuality: 0.3,
  mimeType: 'image/jpeg',
};

export async function compressImage(
  source: Blob,
  options: CompressionOptions = DEFAULT_COMPRESSION,
): Promise<Blob> {
  const img = await loadImage(source);
  const canvas = resize(img, options.maxDimension);
  let blob = await canvasToBlob(canvas, options.initialQuality, options.mimeType);
  if (blob.size > options.maxBytes) {
    blob = await canvasToBlob(canvas, options.fallbackQuality, options.mimeType);
  }
  if (blob.size > options.maxBytes) {
    blob = await canvasToBlob(canvas, options.floorQuality, options.mimeType);
  }
  if (blob.size > options.maxBytes) {
    throw new Error('FILE_TOO_LARGE_AFTER_COMPRESSION');
  }
  return blob;
}

async function loadImage(blob: Blob): Promise<HTMLImageElement> {
  const url = URL.createObjectURL(blob);
  try {
    const img = new Image();
    img.src = url;
    await img.decode();
    return img;
  } finally {
    URL.revokeObjectURL(url);
  }
}

function resize(img: HTMLImageElement, maxDimension: number): HTMLCanvasElement {
  const { naturalWidth: w, naturalHeight: h } = img;
  const scale = Math.min(1, maxDimension / Math.max(w, h));
  const canvas = document.createElement('canvas');
  canvas.width = Math.round(w * scale);
  canvas.height = Math.round(h * scale);
  const ctx = canvas.getContext('2d')!;
  ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
  return canvas;
}

function canvasToBlob(canvas: HTMLCanvasElement, quality: number, mime: string): Promise<Blob> {
  return new Promise((resolve, reject) => {
    canvas.toBlob(
      (blob) => (blob ? resolve(blob) : reject(new Error('CANVAS_TO_BLOB_FAILED'))),
      mime,
      quality,
    );
  });
}

export async function generateThumbnail(
  source: Blob,
  size: number = 128,
): Promise<Blob> {
  const img = await loadImage(source);
  const canvas = document.createElement('canvas');
  canvas.width = size;
  canvas.height = size;
  const ctx = canvas.getContext('2d')!;
  // Cover-fit: scale to fill, crop to square
  const scale = Math.max(size / img.naturalWidth, size / img.naturalHeight);
  const w = img.naturalWidth * scale;
  const h = img.naturalHeight * scale;
  const x = (size - w) / 2;
  const y = (size - h) / 2;
  ctx.drawImage(img, x, y, w, h);
  return canvasToBlob(canvas, 0.8, 'image/jpeg');
}

export interface RetryOptions {
  maxAttempts: number;        // 3
  delays: number[];           // [1000, 2000, 4000]
}

export const DEFAULT_RETRY: RetryOptions = {
  maxAttempts: 3,
  delays: [1000, 2000, 4000],
};

export async function uploadWithRetry(
  upload: () => Observable<PhotoResponse>,
  options: RetryOptions = DEFAULT_RETRY,
): Promise<PhotoResponse> {
  let lastError: unknown;
  for (let attempt = 0; attempt < options.maxAttempts; attempt++) {
    try {
      return await firstValueFrom(upload());
    } catch (err) {
      lastError = err;
      if (attempt < options.maxAttempts - 1) {
        await sleep(options.delays[attempt]);
      }
    }
  }
  throw lastError;
}

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}
```

**`<acceptance_criteria>`**
- `compressImage` reduces 8MB input to ≤500KB (test with synthetic 8MB PNG)
- `compressImage` uses ROADMAP quality ladder (0.8 → 0.6 → 0.3)
- `generateThumbnail` produces 128×128 JPEG
- `uploadWithRetry` retries on transient HTTP errors (not on 4xx validation)
- Canvas redraw strips EXIF (verified by `exifr` test in Task 9)

  currentSrc = computed(() => {
    const photo = this.currentPhoto();
    return photo ? `/api/v1/photos/${photo.id}` : '';
  });

  constructor() {
    effect(() => {
      // Reset index when photos array changes
      const start = this.startIndex();
      if (start >= 0 && start < this.photos().length) {
        this.currentIndex.set(start);
      }
    });
  }

  close(): void { this.closed.emit(); }
  prev(): void {
    this.currentIndex.update((i) => (i === 0 ? this.photos().length - 1 : i - 1));
  }
  next(): void {
    this.currentIndex.update((i) => (i === this.photos().length - 1 ? 0 : i + 1));
  }
  onBackdropClick(): void { this.close(); }
  onDelete(): void {
    const photo = this.currentPhoto();
    if (photo) this.photoDeleted.emit(photo.id);
  }
}

  async startCamera(): Promise<void> {
    try {
      this.stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: 'environment' },
        audio: false,
      });
      const video = document.querySelector('video');
      if (video) {
        video.srcObject = this.stream;
        this.videoElement = video;
      }
      this.status.set('previewing');
    } catch (err) {
      this.toast.error('Camera access denied. Switching to upload.');
      this.switchMode('upload');
    }
  }

  stopCamera(): void {
    if (this.stream) {
      this.stream.getTracks().forEach((t) => t.stop());
      this.stream = undefined;
    }
  }

  async capture(): Promise<void> {
    if (!this.videoElement) return;
    this.status.set('capturing');
    const blob = await this.captureFrame();
    setTimeout(() => this.processBlob(blob), 200); // flash animation
  }

  private async captureFrame(): Promise<Blob> {
    const canvas = document.createElement('canvas');
    canvas.width = this.videoElement!.videoWidth;
    canvas.height = this.videoElement!.videoHeight;
    const ctx = canvas.getContext('2d')!;
    ctx.drawImage(this.videoElement!, 0, 0);
    return new Promise((resolve) => {
      canvas.toBlob((blob) => resolve(blob!), 'image/jpeg', 0.95);
    });
  }

  onFileSelected(event: Event): void {
    const input = event.target as HTMLInputElement;
    const file = input.files?.[0];
    if (!file) return;
    if (!this.isValidType(file)) {
      this.toast.error('File type not supported. Use JPEG, PNG, or HEIC.');
      return;
    }
    if (file.size > 8 * 1024 * 1024) {
      this.toast.error('Photo too large to upload. Try a smaller photo.');
      return;
    }
    this.selectedFile.set(file);
  }

  onDragOver(event: DragEvent): void { event.preventDefault(); }
  onDrop(event: DragEvent): void {
    event.preventDefault();
    const file = event.dataTransfer?.files[0];
    if (file) this.selectedFile.set(file);
  }

  removeFile(event: Event): void {
    event.stopPropagation();
    this.selectedFile.set(null);
  }

  isValidType(file: File): boolean {
    return ['image/jpeg', 'image/png', 'image/heic'].includes(file.type);
  }

  switchMode(mode: Mode): void {
    if (mode === 'upload') this.stopCamera();
    this.mode.set(mode);
  }

  async upload(): Promise<void> {
    const file = this.selectedFile() ?? await this.captureFrame();
    await this.processBlob(file);
  }

  private async processBlob(blob: Blob): Promise<void> {
    this.status.set('compressing');
    let compressed: Blob;
    try {
      compressed = await compressImage(blob);
    } catch (err) {
      this.toast.error('Photo too large to upload. Try a smaller photo.');
      this.status.set('error');
      return;
    }

    this.status.set('uploading');
    this.retryAttempt.set(0);
    try {
      const photo = await uploadWithRetry(() =>
        this.photosApi.upload(compressed, this.entityType(), this.entityId()),
      );
      this.toast.success('Photo uploaded');
      this.photoUploaded.emit(photo);
      this.close();
    } catch (err) {
      this.toast.error('Upload failed. Try again?');
      this.status.set('error');
    }
  }

  close(): void {
    this.stopCamera();
    this.selectedFile.set(null);
    this.status.set('idle');
    this.closed.emit();
  }

  closed = output<void>();
}
