```typescript
export type EntryState =
  | 'entered_with_consent'
  | 'entered_override'
  | 'gatehouse_only'
  | 'denied'
  | 'entered_without_consent';

export type SubjectType = 'dweller' | 'visitor' | 'service-provider' | 'vehicle';
export type OverrideReason = 'emergency' | 'vouched';

export interface CreateEntryLogRequest {
  entryState: EntryState;
  subjectType: SubjectType;
  subjectName?: string;
  subjectDocument?: string;
  photoId?: string;
  overrideReason?: OverrideReason;
}

export interface EntryLogResponse {
  id: string;
  tenantId: string;
  entryState: EntryState;
  overrideReason?: OverrideReason;
  photoId?: string;
  subjectType: SubjectType;
  subjectName?: string;
  subjectDocument?: string;
  performedByProfileId?: string;
  recordedAt: string; // ISO-8601 with milliseconds
}

export interface AuditFilters {
  entryState?: EntryState;
  subjectType?: SubjectType;
  fromUtc?: string;
  toUtc?: string;
  skip: number;
  take: number;
}

@Injectable({ providedIn: 'root' })
export class EntryLogService {
  private http = inject(HttpClient);

  create(request: CreateEntryLogRequest): Observable<EntryLogResponse> {
    return this.http.post<EntryLogResponse>('/api/v1/entry-log', request);
  }

  list(filters: AuditFilters): Observable<EntryLogResponse[]> {
    let params = new HttpParams()
      .set('skip', filters.skip.toString())
      .set('take', filters.take.toString());
    if (filters.entryState) params = params.set('entryState', filters.entryState);
    if (filters.subjectType) params = params.set('subjectType', filters.subjectType);
    if (filters.fromUtc) params = params.set('fromUtc', filters.fromUtc);
    if (filters.toUtc) params = params.set('toUtc', filters.toUtc);
    return this.http.get<EntryLogResponse[]>('/api/v1/entry-log', { params });
  }

  export(filters: AuditFilters): Observable<Blob> {
    let params = new HttpParams();
    if (filters.entryState) params = params.set('entryState', filters.entryState);
    if (filters.subjectType) params = params.set('subjectType', filters.subjectType);
    if (filters.fromUtc) params = params.set('fromUtc', filters.fromUtc);
    if (filters.toUtc) params = params.set('toUtc', filters.toUtc);
    return this.http.get('/api/v1/entry-log/export', {
      params,
      responseType: 'blob',
    });
  }
}
```

export type SubjectCategory = 'dwellers' | 'visitors' | 'service-providers' | 'vehicles';

export interface ConsentPolicyResponse {
  id: string;
  tenantId: string;
  subjectCategory: SubjectCategory;
  photoRequired: boolean;
  dwellTimeLimitMinutes?: number;
  updatedByProfileId?: string;
  createdAtUtc: string;
  updatedAtUtc?: string;
}

export interface UpdateConsentPolicyRequest {
  subjectCategory: SubjectCategory;
  photoRequired: boolean;
  dwellTimeLimitMinutes?: number;
}

@Injectable({ providedIn: 'root' })
export class ConsentPolicyService {
  private http = inject(HttpClient);

  // Backend only has GET /consent-policy/{category} — call 4× in parallel.
  // Forward-compatible: when backend adds GET /consent-policy (list), swap to single call.
  getAll(): Observable<ConsentPolicyResponse[]> {
    const categories: SubjectCategory[] = ['dwellers', 'visitors', 'service-providers', 'vehicles'];
    const requests = categories.map((cat) =>
      this.http.get<ConsentPolicyResponse>(`/api/v1/consent-policy/${cat}`).pipe(
        catchError(() => of(null)), // missing policy → null
      ),
    );
    return forkJoin(requests).pipe(
      map((results) => results.filter((r): r is ConsentPolicyResponse => r !== null)),
    );
  }

  update(request: UpdateConsentPolicyRequest): Observable<ConsentPolicyResponse> {
    return this.http.put<ConsentPolicyResponse>('/api/v1/consent-policy', request);
  }

  updateAll(policies: UpdateConsentPolicyRequest[]): Observable<ConsentPolicyResponse[]> {
    const requests = policies.map((p) => this.update(p));
    return forkJoin(requests);
  }
}

export class CeEntryWorkflowComponent {
  private entryLogService = inject(EntryLogService);
  private consentPolicyService = inject(ConsentPolicyService);
  private toast = inject(ToastService);

  open = input<boolean>(false);
  defaultCategory = input<SubjectType>('visitor');

  entryLogged = output<EntryLogResponse>();
  closed = output<void>();

  step = signal<'tiles' | 'subject-info'>('tiles');
  pendingState = signal<EntryState | null>(null);
  showCapture = signal(false);
  showOverrideReason = signal(false);
  loading = signal(false);
  subjectName = signal('');
  subjectDocument = signal('');
  selectedCategory = signal<SubjectType>('visitor');
  photoId = signal<string | null>(null);
  pendingOverrideReason = signal<OverrideReason | null>(null);

  async onTileTap(action: 'register' | 'denied' | 'gatehouse' | 'override'): Promise<void> {
    if (action === 'register') {
      this.pendingState.set('entered_with_consent');
      // Check policy: if photo required, open camera; else go to subject info
      const policies = await firstValueFrom(this.consentPolicyService.getAll());
      const policy = policies.find((p) => p.subjectCategory === 'visitors');
      if (policy?.photoRequired) {
        this.selectedCategory.set('visitor');
        this.showCapture.set(true);
      } else {
        this.step.set('subject-info');
      }
    } else if (action === 'denied') {
      this.pendingState.set('denied');
      this.step.set('subject-info');
    } else if (action === 'gatehouse') {
      this.pendingState.set('gatehouse_only');
      this.selectedCategory.set('service-provider');
      this.step.set('subject-info');
    } else if (action === 'override') {
      this.pendingState.set('entered_override');
      this.showOverrideReason.set(true);
    }
  }

  back(): void {
    this.step.set('tiles');
    this.subjectName.set('');
    this.subjectDocument.set('');
    this.photoId.set(null);
  }

  close(): void {
    this.reset();
    this.closed.emit();
  }

  private reset(): void {
    this.step.set('tiles');
    this.pendingState.set(null);
    this.showCapture.set(false);
    this.showOverrideReason.set(false);
    this.subjectName.set('');
    this.subjectDocument.set('');
    this.photoId.set(null);
    this.pendingOverrideReason.set(null);
    this.loading.set(false);
  }

  onPhotoUploaded(photo: any): void {
    this.photoId.set(photo.id);
    this.showCapture.set(false);
    this.step.set('subject-info');
  }

  onCaptureClosed(): void {
    if (!this.photoId()) {
      // User cancelled camera — go back to tiles
      this.showCapture.set(false);
      this.reset();
    }
  }

  onOverrideReason(reason: OverrideReason): void {
    this.pendingOverrideReason.set(reason);
    this.showOverrideReason.set(false);
    this.step.set('subject-info');
  }

  async continue(): Promise<void> {
    if (!this.pendingState()) return;
    this.loading.set(true);
    try {
      const request: CreateEntryLogRequest = {
        entryState: this.pendingState()!,
        subjectType: this.selectedCategory(),
        subjectName: this.subjectName() || undefined,
        subjectDocument: this.subjectDocument() || undefined,
        photoId: this.photoId() || undefined,
        overrideReason: this.pendingOverrideReason() || undefined,
      };
      const entry = await firstValueFrom(this.entryLogService.create(request));
      this.toast.success('Entry logged');
      this.entryLogged.emit(entry);
      this.close();
    } catch (err) {
      this.toast.error('Failed to log entry. Retry?');
      this.loading.set(false);
    }
  }
}

export class ConsentPolicyEditorPage implements OnInit {
  private consentPolicyService = inject(ConsentPolicyService);
  private toast = inject(ToastService);

  categories = [
    { value: 'dwellers' as SubjectCategory, label: 'Dwellers', description: 'Residents living in the condominium' },
    { value: 'visitors' as SubjectCategory, label: 'Visitors', description: 'Visitors and guests' },
    { value: 'service-providers' as SubjectCategory, label: 'Service-Providers', description: 'Service providers and contractors' },
    { value: 'vehicles' as SubjectCategory, label: 'Vehicles', description: 'Vehicles entering the gate' },
  ];

  policies = signal<ConsentPolicyResponse[]>([]);
  draft = signal<Map<SubjectCategory, boolean>>(new Map());
  loading = signal(false);
  saving = signal(false);

  isRequired = (cat: SubjectCategory) => this.draft().get(cat) ?? false;
  dirty = computed(() => {
    for (const cat of this.categories) {
      const current = this.policies().find((p) => p.subjectCategory === cat.value)?.photoRequired ?? false;
      const draft = this.draft().get(cat.value) ?? false;
      if (current !== draft) return true;
    }
    return false;
  });

  lastUpdatedText = computed(() => {
    const latest = this.policies().reduce((max, p) => {
      const t = p.updatedAtUtc ? new Date(p.updatedAtUtc).getTime() : 0;
      return t > max ? t : max;
    }, 0);
    if (!latest) return 'Never updated';
    return `Last updated ${new Date(latest).toLocaleString()}`;
  });

  ngOnInit(): void {
    this.load();
  }

  private async load(): Promise<void> {
    this.loading.set(true);
    try {
      const policies = await firstValueFrom(this.consentPolicyService.getAll());
      this.policies.set(policies);
      const draft = new Map<SubjectCategory, boolean>();
      for (const cat of this.categories) {
        const existing = policies.find((p) => p.subjectCategory === cat.value);
        draft.set(cat.value, existing?.photoRequired ?? false);
      }
      this.draft.set(draft);
    } catch {
      this.toast.error('Failed to load consent policy');
    } finally {
      this.loading.set(false);
    }
  }

  onToggle(cat: SubjectCategory, value: boolean): void {
    const draft = new Map(this.draft());
    draft.set(cat, value);
    this.draft.set(draft);
  }

  async save(): Promise<void> {
    this.saving.set(true);
    try {
      const updates = this.categories
        .filter((cat) => {
          const current = this.policies().find((p) => p.subjectCategory === cat.value)?.photoRequired ?? false;
          return current !== (this.draft().get(cat.value) ?? false);
        })
        .map((cat) => ({
          subjectCategory: cat.value,
          photoRequired: this.draft().get(cat.value) ?? false,
        }));

      const results = await firstValueFrom(this.consentPolicyService.updateAll(updates));
      this.policies.set([...this.policies(), ...results]);
      this.toast.success('Policy updated');
    } catch {
      this.toast.error('Failed to save policy');
    } finally {
      this.saving.set(false);
    }
  }
}
